- **uConnect data/voice** -- SQL (native query)
  ```sql
  SELECT
      m.TENANT,
      DATE_TRUNC('MONTH', m.ACTIVATION_DATE) AS ACTIVATION_MONTH,
      u.USAGE_DATE::DATE AS USAGE_DATE,
      DATE_TRUNC('MONTH', u.USAGE_DATE) AS USAGE_MONTH,
      DATEDIFF(
          'month',
          DATE_TRUNC('MONTH', m.ACTIVATION_DATE),
          DATE_TRUNC('MONTH', u.USAGE_DATE)
      ) AS AGING_MONTH,
      SUM(u.MEGS_USED) AS TOTAL_MEGS_USED,
      SUM(u.MINUTES_USED) AS TOTAL_MINUTES_USED,
      SUM(u.SMS) AS TOTAL_SMS,
      SUM(u.MMS) AS TOTAL_MMS,
      SUM(u.USSD) AS TOTAL_USSD
  FROM UCONNECT_DW.ANALYTICS.VW_UC_USAGE AS u
  LEFT JOIN UCONNECT_DW.ANALYTICS.UCONNECT_MAY_MERGE AS m
      ON m.ACCOUNT_NUMBER = u.ACCOUNT_NUMBER
  WHERE
      CAST(u.USAGE_DATE AS DATE) >= CAST(CURRENT_TIMESTAMP AS DATE) - 970
      AND m.MASTER_TENANT = 'uConnect'
  GROUP BY
      m.TENANT,
      DATE_TRUNC('MONTH', m.ACTIVATION_DATE),
      u.USAGE_DATE::DATE,
      DATE_TRUNC('MONTH', u.USAGE_DATE),
      DATEDIFF(
          'month',
          DATE_TRUNC('MONTH', m.ACTIVATION_DATE),
          DATE_TRUNC('MONTH', u.USAGE_DATE)
      )
  ```

## OKR scorecard
- **7 DAY SUBSCRIPTIONS** -- SQL (native query)
  ```sql
  WITH 
  AppData AS (
      SELECT
          CAST(ACTIVATIONDATE AS DATE) AS sales_date,
          'App' AS sales_channel,
          'App' AS CAMPAIGNNAME,
          MANDATEREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_UCONNECT_SUBSCRIPTIONS
      WHERE MANDATETYPE = 'App'
        AND COALESCE(POSITION('RETAIL_PROMOTIONS' IN EXTERNALREFERENCE), 0) = 0
  ),
  OtherSales AS (
      SELECT
          CAST(SALESDATE AS DATE) AS sales_date,
          CASE
              WHEN CAMPAIGNNAME IN ('Uconnect Upsell', 'UConnect Triplesave')
                   OR CAMPAIGNNAME ILIKE '%Breakfree%' THEN 'Telesales'
              WHEN CAMPAIGNNAME IN ('DigiM Resells')
                   OR CAMPAIGNNAME ILIKE '%Mobile Store%' THEN 'Inbound'
              WHEN CAMPAIGNNAME = 'Digital UConnect Upsell' THEN 'Whatsapp'
              ELSE 'Other'
          END AS sales_channel,
          CAMPAIGNNAME,
          ORDERREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_SILVER_SURFER_SALES_SIM_INFO
      WHERE SALESDATE >= DATEADD(day, -7, CURRENT_DATE())   -- last 7 days
  ),
  RetailSales AS (
      SELECT
          CAST(ACTIVATIONDATE AS DATE) AS sales_date,
          'Retail' AS sales_channel,
          'Uconnect VAS Instore' AS CAMPAIGNNAME,
          MANDATEREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_UCONNECT_SUBSCRIPTIONS
      WHERE EXTERNALREFERENCE ILIKE '%RETAIL_PROMOTIONS%'
  ),
  AllSales AS (
      SELECT * FROM AppData
      UNION ALL
      SELECT * FROM OtherSales
      UNION ALL
      SELECT * FROM RetailSales
  ),
  YesterdayAgg AS (
      SELECT
          sales_channel,
          CAMPAIGNNAME,
          CASE 
              WHEN sales_channel IN ('App', 'Retail') THEN COUNT(DISTINCT sale_id)
              ELSE COUNT(*)
          END AS sales_yesterday
      FROM AllSales
      WHERE sales_date = CURRENT_DATE - 1
      GROUP BY sales_channel, CAMPAIGNNAME
  ),
  Last7Agg AS (
      SELECT
          sales_channel,
          CAMPAIGNNAME,
          CASE
              WHEN sales_channel IN ('App', 'Retail') THEN COUNT(DISTINCT sale_id)
              ELSE COUNT(*)
          END AS row_count_last7
      FROM AllSales
      WHERE sales_date BETWEEN CURRENT_DATE - 7 AND CURRENT_DATE - 1
      GROUP BY sales_channel, CAMPAIGNNAME
  )
  SELECT
      COALESCE(y.sales_channel, l.sales_channel) AS sales_channel,
      COALESCE(y.CAMPAIGNNAME, l.CAMPAIGNNAME) AS CAMPAIGNNAME,
      COALESCE(y.sales_yesterday, 0) AS sales_yesterday,
      CASE
          WHEN COALESCE(y.sales_channel, l.sales_channel) IN ('App', 'Retail')
              THEN ROUND(l.row_count_last7 / 7.0, 2)
          ELSE ROUND(l.row_count_last7 / 5.0, 2)
      END AS last7_avg
  FROM Last7Agg l
  LEFT JOIN YesterdayAgg y
      ON y.sales_channel = l.sales_channel
     AND y.CAMPAIGNNAME = l.CAMPAIGNNAME
  ORDER BY sales_channel, CAMPAIGNNAME
  ```

- **Retail Trading Stores L7Days and Yesterday** -- SQL (native query)
  ```sql
  WITH RETAIL_TRADING_TENANTS_L7DAYS AS (
  
  SELECT  RETAIL_CHANNEL,
          COUNT(DISTINCT TENANT) AS UNIQUE_TRADING_TENANTS_L7DAYS
  FROM UCONNECT_DW.ANALYTICS.UCONNECT_MAY_MERGE AS M
  LEFT JOIN UCONNECT_DW.ANALYTICS.VW_UCONNECT_RETAIL_GROUPS AS RG
      ON RG.COMPANYNAME = M.TENANT
  WHERE COMPANYNAME IS NOT NULL
    AND M.ACTIVATION_DATE >= CURRENT_DATE - INTERVAL '7 DAY'
    AND M.ACTIVATION_DATE < CURRENT_DATE
  GROUP BY RETAIL_CHANNEL
  
  ),
  
  RETAIL_TRADING_TENANTS_YESTERDAY AS (
  SELECT  RETAIL_CHANNEL,
          COUNT(DISTINCT TENANT) AS UNIQUE_TRADING_TENANTS_YESTERDAY
  FROM UCONNECT_DW.ANALYTICS.UCONNECT_MAY_MERGE AS M
  LEFT JOIN UCONNECT_DW.ANALYTICS.VW_UCONNECT_RETAIL_GROUPS AS RG
      ON RG.COMPANYNAME = M.TENANT
  WHERE COMPANYNAME IS NOT NULL
    AND M.ACTIVATION_DATE = CURRENT_DATE - 1
  GROUP BY RETAIL_CHANNEL
  
  )
  
  SELECT  COALESCE(L7.RETAIL_CHANNEL, Y.RETAIL_CHANNEL) AS RETAIL_CHANNEL,
          L7.UNIQUE_TRADING_TENANTS_L7DAYS,
          Y.UNIQUE_TRADING_TENANTS_YESTERDAY
  FROM RETAIL_TRADING_TENANTS_L7DAYS L7
  FULL OUTER JOIN RETAIL_TRADING_TENANTS_YESTERDAY Y
      ON L7.RETAIL_CHANNEL = Y.RETAIL_CHANNEL
  ORDER BY RETAIL_CHANNEL
  ```

- **7 DAY SUBSCRIPTIONS** -- SQL (native query)
  ```sql
  WITH 
  AppData AS (
      SELECT
          CAST(ACTIVATIONDATE AS DATE) AS sales_date,
          'App' AS sales_channel,
          'App' AS CAMPAIGNNAME,
          MANDATEREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_UCONNECT_SUBSCRIPTIONS
      WHERE MANDATETYPE = 'App'
        AND COALESCE(POSITION('RETAIL_PROMOTIONS' IN EXTERNALREFERENCE), 0) = 0
  ),
  OtherSales AS (
      SELECT
          CAST(SALESDATE AS DATE) AS sales_date,
          CASE
              WHEN CAMPAIGNNAME IN ('Uconnect Upsell', 'UConnect Triplesave')
                   OR CAMPAIGNNAME ILIKE '%Breakfree%' THEN 'Telesales'
              WHEN CAMPAIGNNAME IN ('DigiM Resells')
                   OR CAMPAIGNNAME ILIKE '%Mobile Store%' THEN 'Inbound'
              WHEN CAMPAIGNNAME = 'Digital UConnect Upsell' THEN 'Whatsapp'
              ELSE 'Other'
          END AS sales_channel,
          CAMPAIGNNAME,
          ORDERREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_SILVER_SURFER_SALES_SIM_INFO
      WHERE SALESDATE >= DATEADD(day, -7, CURRENT_DATE())   -- last 7 days
  ),
  RetailSales AS (
      SELECT
          CAST(ACTIVATIONDATE AS DATE) AS sales_date,
          'Retail' AS sales_channel,
          'Uconnect VAS Instore' AS CAMPAIGNNAME,
          MANDATEREFERENCE AS sale_id
      FROM UCONNECT_DW.ANALYTICS.VW_UCONNECT_SUBSCRIPTIONS
      WHERE EXTERNALREFERENCE ILIKE '%RETAIL_PROMOTIONS%'
  ),
  AllSales AS (
      SELECT * FROM AppData
      UNION ALL
      SELECT * FROM OtherSales
      UNION ALL
      SELECT * FROM RetailSales
  ),
  YesterdayAgg AS (
      SELECT
          sales_channel,
          CAMPAIGNNAME,
          CASE 
              WHEN sales_channel IN ('App', 'Retail') THEN COUNT(DISTINCT sale_id)
              ELSE COUNT(*)
          END AS sales_yesterday
      FROM AllSales
      WHERE sales_date = CURRENT_DATE - 1
      GROUP BY sales_channel, CAMPAIGNNAME
  ),
  Last7Agg AS (
      SELECT
          sales_channel,
          CAMPAIGNNAME,
          CASE
              WHEN sales_channel IN ('App', 'Retail') THEN COUNT(DISTINCT sale_id)
              ELSE COUNT(*)
          END AS row_count_last7
      FROM AllSales
      WHERE sales_date BETWEEN CURRENT_DATE - 7 AND CURRENT_DATE - 1
      GROUP BY sales_channel, CAMPAIGNNAME
  )
  SELECT
      COALESCE(y.sales_channel, l.sales_channel) AS sales_channel,
      COALESCE(y.CAMPAIGNNAME, l.CAMPAIGNNAME) AS CAMPAIGNNAME,
      COALESCE(y.sales_yesterday, 0) AS sales_yesterday,
      CASE
          WHEN COALESCE(y.sales_channel, l.sales_channel) IN ('App', 'Retail')
              THEN ROUND(l.row_count_last7 / 7.0, 2)
          ELSE ROUND(l.row_count_last7 / 5.0, 2)
      END AS last7_avg
  FROM Last7Agg l
  LEFT JOIN YesterdayAgg y
      ON y.sales_channel = l.sales_channel
     AND y.CAMPAIGNNAME = l.CAMPAIGNNAME
  ORDER BY sales_channel, CAMPAIGNNAME
  ```

## Sales trends
- **Sales L7 Day Avg** -- SQL (native query)
  ```sql
  WITH date_spine AS (
      -- Need extra history to compute the first 7-day window for the first visible day
      -- 30 visible days + 7 lookback + buffer
      SELECT DATEADD(day, -seq4(), CURRENT_DATE()) AS dt
      FROM TABLE(GENERATOR(ROWCOUNT => 45))
  ),
  
  tenant_list AS (
      -- Tenants to include (restricting to those with any sales in the extended window is fine)
      SELECT DISTINCT TENANT
      FROM UCONNECT_DW.ANALYTICS.UCONNECT_MAY_MERGE
      WHERE CAST(ACTIVATION_DATE AS DATE) >= DATEADD(day, -45, CURRENT_DATE())
        AND CAST(ACTIVATION_DATE AS DATE) <  CURRENT_DATE()
  ),
  
  tenant_date_spine AS (
      -- One row per tenant per day
      SELECT t.TENANT, d.dt
      FROM tenant_list t
      CROSS JOIN date_spine d
  ),
  
  daily_sales AS (
      -- Daily sales count per tenant
      SELECT
          TENANT,
          CAST(ACTIVATION_DATE AS DATE) AS dt,
          COUNT(ACCOUNT_NUMBER)   AS sales_cnt
      FROM UCONNECT_DW.ANALYTICS.UCONNECT_MAY_MERGE
      WHERE CAST(ACTIVATION_DATE AS DATE) >= DATEADD(day, -45, CURRENT_DATE())
        AND CAST(ACTIVATION_DATE AS DATE) <  CURRENT_DATE()   -- exclude today
      GROUP BY 1, 2
  ),
  
  base AS (
      -- Fill missing days with 0
      SELECT
          td.TENANT,
          td.dt,
          COALESCE(ds.sales_cnt, 0) AS sales_cnt
      FROM tenant_date_spine td
      LEFT JOIN daily_sales ds
          ON ds.TENANT = td.TENANT
         AND ds.dt     = td.dt
  )
  
  SELECT
      TENANT,
      dt,
  
      -- Rolling 7-day average excluding dt:
      -- sums dt-7..dt-1 (7 days) / 7
      SUM(sales_cnt) OVER (
          PARTITION BY TENANT
          ORDER BY dt
          ROWS BETWEEN 7 PRECEDING AND 1 PRECEDING
      ) / 7.0 AS rolling_7day_avg
  
  FROM base
  WHERE dt >= DATEADD(day, -30, CURRENT_DATE())  -- last 30 days
    AND dt <=  CURRENT_DATE()                     -- exclude today as a displayed day (optional)
  ORDER BY TENANT, dt
  ```

- **Quality of sales Last 7 days** -- SQL (native query)
  ```sql
  -- QoS & ROS last 7 days with filtering
  WITH DATESERIES AS (
      SELECT CURRENT_DATE() - INTERVAL '400 DAYS' AS "DATE"
      UNION ALL
      SELECT "DATE" + INTERVAL '1 DAY'
      FROM DATESERIES
      WHERE "DATE" + INTERVAL '1 DAY' <= CURRENT_DATE()
  ),
  
  TENANTS AS (
      SELECT DISTINCT TENANT
      FROM "UCONNECT_DW"."ANALYTICS"."UCONNECT_MAY_MERGE"
      WHERE MASTER_TENANT = 'uConnect'
  ),
  
  DATESERIES_TENANT AS (
      SELECT 
          t.TENANT,
          d."DATE"
      FROM TENANTS t, DATESERIES d
  ),
  
  DISTINCT_SALES AS (
      SELECT
          M.ACCOUNT_NUMBER,
          M.TENANT,
          M.ACTIVATION_DATE
      FROM "UCONNECT_DW"."ANALYTICS"."UCONNECT_MAY_MERGE" AS M
      WHERE M.MASTER_TENANT = 'uConnect'
      GROUP BY M.ACCOUNT_NUMBER, M.TENANT, M.ACTIVATION_DATE
  ),
  
  SALESDATA AS (
      SELECT
          ds_t."DATE" AS REFERENCE_DATE,
          ds_t.TENANT,
          COUNT(DISTINCT M.ACCOUNT_NUMBER) AS SALES_COUNT
      FROM DATESERIES_TENANT AS ds_t
      JOIN DISTINCT_SALES AS M
          ON M.ACTIVATION_DATE BETWEEN DATEADD(DAY, -6, ds_t."DATE") AND ds_t."DATE"
          AND M.TENANT = ds_t.TENANT
      WHERE ds_t."DATE" >= CURRENT_DATE() - INTERVAL '60 DAYS'
      GROUP BY ds_t."DATE", ds_t.TENANT
  ),
  
  USAGEDATA AS (
      SELECT
          ds_t.TENANT,
          ds_t."DATE" AS REFERENCE_DATE,
          COUNT(DISTINCT M.ACCOUNT_NUMBER) AS USAGE_COUNT
      FROM "UCONNECT_DW"."ANALYTICS"."UCONNECT_MAY_MERGE" AS M
      JOIN "UCONNECT_DW"."ANALYTICS"."VW_UC_USAGE" AS U
          ON M.ACCOUNT_NUMBER = U.ACCOUNT_NUMBER
      JOIN DATESERIES_TENANT AS ds_t
          ON M.ACTIVATION_DATE BETWEEN DATEADD(DAY, -6, ds_t."DATE") AND ds_t."DATE"
          AND M.TENANT = ds_t.TENANT
      WHERE 
          M.MASTER_TENANT = 'uConnect'
          AND U.USAGE_DATE BETWEEN DATEADD(DAY, -6, ds_t."DATE") AND ds_t."DATE"
      GROUP BY ds_t."DATE", ds_t.TENANT
      HAVING (
          SUM(COALESCE(U.MINUTES_USED, 0)) +
          SUM(COALESCE(U.MEGS_USED, 0)) +
          SUM(COALESCE(U.SMS, 0)) +
          SUM(COALESCE(U.USSD, 0)) +
          SUM(COALESCE(U.MMS, 0))
      ) > 0
  ),
  
  FINALDATA AS (
      SELECT
          S.REFERENCE_DATE,
          S.TENANT,
          S.SALES_COUNT,
          COALESCE(U.USAGE_COUNT, 0) AS USAGE_COUNT,
          ROUND(S.SALES_COUNT / 6, 2) AS ROS_7_DAYS,
          CASE
              WHEN S.SALES_COUNT > 0 THEN ROUND(COALESCE(U.USAGE_COUNT, 0) / S.SALES_COUNT, 2)
              ELSE 0
          END AS USAGE_PERCENTAGE
      FROM SALESDATA AS S
      LEFT JOIN USAGEDATA AS U
          ON S.REFERENCE_DATE = U.REFERENCE_DATE
          AND S.TENANT = U.TENANT
  )
  
  SELECT 
      F.*,
      CASE 
          WHEN F.USAGE_PERCENTAGE < 0.50 THEN 1 
          ELSE 0 
      END AS QOS_Flag,
      ROUND(AVG(F.USAGE_PERCENTAGE) OVER (
          PARTITION BY F.TENANT 
          ORDER BY F.REFERENCE_DATE 
          ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
      ) / 6, 4) AS USAGE_PERCENTAGE_7DAY_AVG_DIV6
  FROM FINALDATA AS F
  WHERE F.SALES_COUNT > 0 AND F.ROS_7_DAYS > 0
  ORDER BY F.TENANT, F.REFERENCE_DATE
  ```
